function t3linkSoftware(uri, name, version, date) {

    } else {

        url += '&ext=.dll';
    }

    document.location = url;

		var name = cname + "=";
		var ca = document.cookie.split(';');
		for ( var i = 0; i < ca.length; i++) {
			var c = ca[i];
			while (c.charAt(0) == ' ')
				c = c.substring(1);
			if (c.indexOf(name) == 0) {
				return c.substring(name.length, c.length);
			}
		}
		return "";
	}

		} else if(failureMsg=="Authentication Failed: Too many login failure attempts, User account has been locked") {

 		} else {
			errorData.innerHTML = failureMsg;
		}

    function cookieCreateCheck() {
      
    	var tessURL = "https://one.tis.toyota.com:443/.well-known/security.txt";  
    	//security vulnerability fix starts    	 
    	var pattern = new RegExp(/[!#$*();|]/); //unacceptable chars
    	var acceptablepattern = new RegExp(/^(http|https):\/\/(\w+:{0,1}\w*)?(\S+)(:[0-9]+)?(\/|\/([\w#!:.?+=&%!\-\/\']))?/);
    	var securityFixEnableFlag = "Y"; 
    	if((securityFixEnableFlag=='Y')&&((pattern.test(tessURL)) || !(acceptablepattern.test(tessURL))))
    	{
    		var urlValue = "/techInfoPortal"+"/resources/jsp/t3error.jsp";    	
			window.location.href = urlValue;
			
			//security vulnerability fix ends	
    	}
    	else{
    		
		var urlValue = decodeURIComponent(decodeURIComponent(tessURL));      
		
		
    	var vinSearchUrl ="https://one.tis.toyota.com/t3Portal/scantool?l=vinSearch";
    	
		if (urlValue != null && urlValue.indexOf("refresh") < 0) {	
    		if(urlValue.indexOf("?") < 0) {
    		  urlValue = urlValue + "?" + "refresh=true"; 
    		} else {
    		  urlValue = urlValue + "&" + "refresh=true";
    		}

    		if(urlValue.indexOf("calibrationValidationCheck") > 0)
    		{
    			var vin =urlValue.split("calibrationValidationCheck")[1];
    			 urlValue = vinSearchUrl + "&vin="+vin;
    		}
    			//alert(urlValue);
    		if(urlValue.indexOf("tis_ecu_mac,") > 0)
    		{
    			var ecuMacKeyUrl ="https://one.tis.toyota.com/t3Portal/scantool?l=tis_ecu_mac&ecuMacKeyAuth=true&ecuMacId=";
    			var ecuMacKeyId =urlValue.split(",")[1];
    			 urlValue = ecuMacKeyUrl+ecuMacKeyId;
    		}
    		
    		if(urlValue.indexOf("tis_search_blank_ecu,") > 0)
    		{
    			var blankEcuUrl ="https://one.tis.toyota.com/t3Portal/scantool?l=tis_search_blank_ecu&blankEcuAuth=true&blankEcuId=";
    			var blankEcuId =urlValue.split(",")[1];
    			 urlValue = blankEcuUrl+blankEcuId;
    		}
    		if(urlValue.indexOf("tis_update_blank_ecu,") > 0)
    		{
    			var blankEcuUrl ="https://one.tis.toyota.com/t3Portal/scantool?l=tis_update_blank_ecu&blankEcuAuth=true&blankEcuId=";
    			var blankEcuId =urlValue.split(",")[1];
    			 urlValue = blankEcuUrl+blankEcuId;
    		}
    		
    		if(urlValue.indexOf("tis_ecu_repro,") > 0)
    		{
    			var ecuReproUrl ="https://one.tis.toyota.com/t3Portal/scantool?l=tis_ecu_repro&ecuMacReproAuth=true&ecuReproId=";
    			var ecuReproId =urlValue.split(",")[1];
    			 urlValue = ecuReproUrl+ecuReproId;
    		}
    		if(urlValue.indexOf('t3.tms.toyota') >=0 || 
    				urlValue.indexOf('one.tis.toyota') >=0  || 
    				urlValue.indexOf('techinfo.toyota') >=0)
    		{
    			window.location.href = urlValue;
    		}
    		else
    		{
    			urlValue = "/techInfoPortal"+"/resources/jsp/t3error.jsp";
    			window.location.href = urlValue;
    		}
    		
    	} 
		else {
    		document.getElementById("pleaseWait-div").style.display = 'none';	
        	document.getElementById("form-div").style.display = 'block';
    	}
    	}
		
	}
