      function initMobileKeyHandler()
      {
        var encodedURI = encodeURIComponent( globals.protocol + "://" + globals.hostname + globals.url );
        // This has to work for all environments
        if ( !globals.mkHostName )
        {
          globals.mkHostName = globals.hostname

        if (globals.mobileKeyUserId && globals.hlte26DecryptEnabled)
        {
          userId = globals.mobileKeyUserId;
        }
        else
        {
          userId = document.getElementById("mobileKeyDeskTopATTUID1").value;
          if ( isMobileDevice() )
          {
            var tmpUserId = document.getElementById('mobileKeyDeviceATTUID1').value;
            if ( tmpUserId != "" )
            {
              userId = tmpUserId;
            }
            else
            {
              document.getElementById('mobileDeviceMobileKeyPassCodeATTUID').style.display = "none";
            }
          }
        }

        var elems = document.getElementsByClassName("mkUserId");
        if ( elems != null )
        {
          for (var i = 0; i < elems.length; i+=1)
          {
            var elem = elems[i];
            switch ( elem.tagName.toLowerCase() )

              case "span":
                   elem.textContent = userId;
                   break;
              case "input":
                   elem.value = userId;
                   break;
              default:
                   haloEAlert ( "changeUserId: unknown element " +  elem.tagName.toLowerCase() );
                   break;
            }
          }
        }

        {
          document.getElementById("GlobalLoginPasswordVerificationATTUID").value =  userId;
          changeVerificationForm ( 'GlobalLoginPassword');
          return false;
        }

        globals.mobileKeyHandler = initMobileKeyHandler();

        try
        {
          return globals.mobileKeyHandler.sendPush();
        }
        catch ( err )
        {
          mobileErrorFrontEnd ( globals.springMessages['system.error' ]  + 'Request failed' + err );
        }
      }

      function getErrorMessage(id, text )
      {
        if ( id )
        {
          switch (id.toLowerCase() )
          {
            case strWinAuthErrorOccured:
              return "Unable to access your Windows account.  Please enter your ATTUID/password below.";
              break;
            case strWinAuthErrorCancelled:
              return "Your log on with your Windows domain account was cancelled.";
              break;
            case strWinAuthErrorWindowsAuthFailed:
              return "Unable to authenticate with your Windows account.  Please enter your ATTUID/password below.";
              break;
            case strWinAuthErrorInvalidDomain:
              return "Unable to access your Windows domain/account.  Please enter your ATTUID/password below.";
              break;
            case strWinAuthErrorCantFindATTUID:
              return "Your Windows account is not linked to an ATTUID.  Please enter your ATTUID/password below.";
              break;
            case strWinAuthErrorIDNotInCSP:
              return "Your Windows account is not linked to a known ATTUID.  Please enter your ATTUID/password below.";
              break;
            case strWinAuthErrorInvalidCSPDomain:
              return "Your Windows account is not linked to an active ATTUID.  Please enter your ATTUID/password below.";
              break;
            case strWinAuthErrorPwdExpired:
              return "Your AT&amp;T Password needs to be changed.  Windows auto-auth is temporarily suspended.  Please enter your ATTUID/password below, after which you will be asked to change your password.";
              break;
            case strWinAuthErrorBlockedAutoAuth:
              return "Windows auto-authentication is disabled for this ATTUID.  To enable, click <a href=\"https://oidc.idp.elogin.att.com/iamportal-prod/accountreset/#/resetpassword?register=yes\">here</a>. Please use your ATTUID/password to log on.  ";
              break;
            case strWinAuthErrorCantAutoAuth:
              return "System error.  Unable to access your Windows account.  Please enter your ATTUID/password.";
              break;
            case strMobileKeyErrorHlte26NotFound:
              return "hlte26 cookie not found";
              break;
            case strMobileKeyErrorRememberMeNotFound:
              return "rememberMe cookie not found";
              break;
            case strMobileKeyErrorExcessiveFailures:
              var msg = "Mobile Key disabled due to excessive failures";
              $('#mobileKey').data ('disabled', msg );
              return msg;
              break;
            case strMobileKeyErrorLoginWithAnotherMethod:
              return "Unable to process Mobile Key request. Login with another method";
              break;
            case strMobileKeyErrorNoDevicesRegistered:
              return "Please make sure you're using the latest version of Mobile Key, retry, or select another logon option.";
              break;
            case strMobileKeyErrorUnableToInitiate:
              return "Please make sure you're using the latest version of Mobile Key, retry, or select another logon option.";
              break;
            case strMobileKeyErrorNoSessionId:
              return "No Session ID";
              break;
            case strMobileKeyErrorStatusDenied:
                var status = getCookie(statusCookieName);
                if (status == "denied")
                {
                  deleteCookie(statusCookieName);
                }
                return "Your request has been denied. Send Mobile Key push again or choose another option.";
              break;
            case strMobileKeyErrorInvalidAction:
              return "Invalid action received";
              break;
            case strMobileKeyErrorInvalidValues:
              return "Invalid values received";
              break;
            case strMobileKeyErrorInvalidParameters:
              return "Invalid parameters received";
              break;
            case strMobileKeyErrorSoftLock:
              return "Soft Lock";
              break;
            case "201.10":
              return "Login incorrect. Make sure you are using your Windows Password (ITSERVICES, INTL, or MX) which is now your one AT&amp;T Password. (" + id + ")";
              break;
            case "205.4":
              redirectToLrrError ( id, text, globals.referenceId, globals.sessionId );
              //should never return!!!
              break;
            case "0x132121f4":  //DROP THROUGH
            case "0x132120c8":
              return "Login incorrect. Please try again. (" + id + ")";
              break;

            default:
              if ( text == null || text == "" )
              {
                return "Login incorrect. Please try again. (" + id + ")";
              }
              else
              {
                return text + " (" + id + ")"
              }
              break;
          }
          return  text + "(" + id + ")" + "<br> message redirect failed as well";
        }
      }
