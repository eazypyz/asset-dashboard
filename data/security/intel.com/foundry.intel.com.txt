    function initiateB2CLogoutToken(type) {
        return new Promise(function (resolve, reject) {
            function authIFrameB2CIntervalWrapper(authIFrame) {
                var authIFrameInterval = setInterval(function () {
                    var authIFrameWindow = authIFrame.contentWindow;
                    if (authIFrameWindow) {
                        try {
                            var authIFrameURL = authIFrameWindow.location.href;

                            if (authIFrameURL) {
                                if (authIFrameURL.includes("#id_token=")) {
                                    clearB2CIFrame(authIFrameInterval, authIFrame);
                                    return resolve({
                                        id: authIFrame.id,
                                        msg: "SUCCESS!" + "--" + authIFrame.id,
                                        code: "SUCCESS",
                                        id_token_hint:authIFrameURL.split("#id_token=")[1]

                                } else if (authIFrameURL.includes("#error=")) {
                                    clearB2CIFrame(authIFrameInterval, authIFrame);
                                    return reject({
                                        id: authIFrame.id,
                                        msg: "FAILURE!" + "--" + authIFrame.id,
                                        code: "FAILURE",
                                    });
                                } else {
                                    authIFrameAttempts++;
                                    if (
                                        authIFrameAttempts ===
                                        AUTH_IFRAME_MAX_ATTEMPTS

                                        clearB2CIFrame(authIFrameInterval, authIFrame);

                                    code: "ERROR",

                            msg: "ERROR - DOM!" + "--" + authIFrame.id,
                            code: "ERROR",

    function initiateAuth(type) {
        return new Promise(function (resolve, reject) {
            function authIFrameIntervalWrapper(authIFrame) {
                let authIFrameInterval = setInterval(function () {
                    let authIFrameWindow = authIFrame.contentWindow;
                    if (authIFrameWindow) {
                        try {
                            let authIFrameURL = authIFrameWindow.location.href;
                            if (authIFrameURL) {
                                if (authIFrameURL.includes("#access_token=")) {
                                    clearIFrame(authIFrameInterval, authIFrame);
                                    return resolve({
                                        id: authIFrame.id,
                                        msg: "SUCCESS!" + "--" + authIFrame.id,
                                        code: "SUCCESS",
                                    });
                                } else if (authIFrameURL.includes("#error=")) {
                                    clearIFrame(authIFrameInterval, authIFrame);
                                    return reject({
                                        id: authIFrame.id,
                                        msg: "FAILURE!" + "--" + authIFrame.id,
                                        code: "FAILURE",
                                    });
                                } else {
                                    authIFrameAttempts++;
                                    if (
                                        authIFrameAttempts ===
                                        AUTH_IFRAME_MAX_ATTEMPTS
                                    ) {
                                        clearIFrame(authIFrameInterval, authIFrame);
                                        return reject({
                                            id: authIFrame.id,
                                            msg:
                                                "ERROR - MAX (TIMING)!" +
                                                "--" +
                                                authIFrame.id,
                                            code: "ERROR",
                                        });
                                    }
                                }
                            } else {
                                clearIFrame(authIFrameInterval, authIFrame);
                                return reject({
                                    id: authIFrame.id,
                                    msg: "ERROR - URL!" + "--" + authIFrame.id,
                                    code: "ERROR",
                                });
                            }
                        } catch (err) {
                            if (authIFrameAttempts === AUTH_IFRAME_MAX_ATTEMPTS) {
                                clearIFrame(authIFrameInterval, authIFrame);
                                return reject({
                                    id: authIFrame.id,
                                    msg:
                                        "ERROR - MAX (CORS)!" +
                                        "--" +
                                        authIFrame.id,
                                    code: "ERROR",
                                });
                            }
                            authIFrameAttempts++;
                        }
                    } else {
                        clearIFrame(authIFrameInterval, authIFrame);
                        return reject({
                            id: authIFrame.id,
                            msg: "ERROR - DOM!" + "--" + authIFrame.id,
                            code: "ERROR",
                        });
                    }
                }, 500);
            }

            var AUTH_IFRAME_MAX_ATTEMPTS = 40;
            var authIFrameAttempts = 0;

            var authURL;
            let tempIsIGHFCall = document.querySelectorAll('#recode50header').length > 0;
            var currentOrigin =  encodeURIComponent(window.location.origin + ((!tempIsIGHFCall && !/mark.intel.com/.test(window.location.hostname)) ? intel.signinsimplifyConfig.authorizationUrl : "/"));
            var redirectURI = currentOrigin;

                window.INTELNAV &&

                    site.host.includes(window.location.host)
                );
            if (matchedSite) {
                redirectURI += matchedSite.redirectURI;
            }

            if (type === "b2c") {
                authURL = decodeURIComponent(
                    intel.signinsimplifyConfig.sessionAuthorizeUrlB2C + redirectURI
                );
            } else {
                authURL = decodeURIComponent(
                    intel.signinsimplifyConfig.sessionAuthorizeUrlB2B + redirectURI
                );
            }

            let authIFrame = createIFrame(type, authURL);
            authIFrameIntervalWrapper(authIFrame);
        });

    function getAuthenticationStatus() {
        return new Promise(function (resolve, reject) {
            initiateAuth("b2c")
                .then(function (b2cSuccessRes) {
                    return resolve(b2cSuccessRes);
                })
                .catch(function (b2cFailureRes) {
                    var isB2BSite =
                        window.INTELNAV &&
                        window.INTELNAV.CIDAAS &&
                        window.INTELNAV.CIDAAS.B2BSites.some((site) =>
                            site.includes(window.location.host)
                        );
                    if (isB2BSite) {
                        initiateAuth("b2b")
                            .then(function (b2bSuccessRes) {
                                return resolve(b2bSuccessRes);
                            })
                            .catch(function (b2bFailureRes) {
                                return reject(b2bFailureRes);
                            });
                    } else {
                        return reject(b2cFailureRes);
                    }
                });
        });
    }

    function checkAEMSession(){
        let isSessionExist = false;
        if(!tempIsIGHF){
            $.ajax({
                url: "/libs/apps/intel/myintel/agsroles.json?validateAEMSession=Y",
                async: false,
                success: function (data) {
                    if (data && data["isValidAEMSession"]) {
                        document.cookie = "IGHFLoggedIn=true;domain=.intel.com;path=/";
                        isLoggedin = true;
                        isSessionExist = true;
                    }
                }
            });
        }
        return isSessionExist;
    }

                        }

            });

<script>
    /**
     * Intel_PardotForm
     */
    (function () {
        const CLASS_NAME_COMPONENT = 'pardot-form';
        const CLASS_NAME_VALIDATION_BLOCK = 'validation-msg-block';
        const CLASS_NAME_COUNTRY_BASED_BLOCK = 'country-based-toggle';
        const CLASS_NAME_COUNTRY_BASED_BLOCK_MATCH = `${CLASS_NAME_COUNTRY_BASED_BLOCK}--matched`;
        const CLASS_NAME_OPTIN_CONSENT_TEXT = 'optin-consent-text';
        const CLASS_NAME_LINK_PRIVACY = 'link-privacy';
        const CLASS_NAME_LINK_COOKIES = 'link-cookies';
        const ATTR_PARDOT_FIELD = 'data-pardot-field';
        const ATTR_CONDITIONAL_REQUIRED = 'data-conditional-required';
        const ID_OPTIN_CONSENT = 'optinConsent';
        const POST_API_URL = 'https://go.intel-foundry.com/l/1109162/2025-11-05/sk2nk4';
        const SESSION_ID_PARDOT = 'intelSubmittedForms';

        const ERROR_MSG_DEFAULT = {
            text: "Required fields",
            select: "Please select a value",
            email: "Please enter a valid email",
            embargoCountry: "Your registration cannot proceed. The materials on this site are subject to U.S. and other applicable export control laws and are not accessible from all locations.",
            checkbox: 'Above consent required for submission'
        };

        const I18N_PROPERTY = {
            linkPrivacy: 'category_forms.optIn.policyNoticeURLHighRisk',
            linkCookies: 'category_forms.optIn.cookiesURLHighRisk',
            text: 'category_forms.optIn.requiredFields',
            select: 'category_forms.optIn.countryError',
            email: 'category_forms.optIn.emailError',
            embargoCountry: 'category_forms.embargoedmodal.embargoed-error-message',
            checkbox: 'category_myintel_profile.chineseConsent.errorMessageConsent'
        };

        const EVENT_FORM_SUBMITTED = 'eloqua.form.submitted';
        const EVENT_FORM_SUBMITTED_SUCCESS = 'eloqua.form.submitted.success';
        const EVENT_FORM_SUBMITTED_FAILED = 'eloqua.form.submitted.failed';
        let eloquaFieldChangeTimer;

        class Intel_PardotForm {
            constructor(form) {
                this.form = form;
                this.pardotFormName = "FoundryAEM_MKA-44977";
                this.requiredFields = this.form.querySelectorAll('[required]');
                this.conditionalRequiredFields = this.form.querySelectorAll(`[${ATTR_CONDITIONAL_REQUIRED}]`);
                this.pardotFields = this.form.querySelectorAll(`[${ATTR_PARDOT_FIELD}]`);
                this.countryField = this.form.querySelector(`[data-field-type="country"]`);
                this.submitBtn = this.form.querySelector('[type="submit"]');
                this.countryToggleBlocks = this.form.querySelectorAll(`.${CLASS_NAME_COUNTRY_BASED_BLOCK}`);
                this.embargoCountries = '';
                this.formStatusSubmitted = false;
                this.setPrivacyCookiesOptinLinks();
                this.setLocaleErrMessages();
                this.getCountriesProperties();
                this.initEvents();
                this.checkSessionSubmittedForms();
            }

            setPrivacyCookiesOptinLinks() {
                const optinConsentText = this.form.querySelectorAll(`.${CLASS_NAME_OPTIN_CONSENT_TEXT}`);
                if (optinConsentText.length < 1) {
                    return;
                }
                if (typeof INTEL !== 'undefined' && INTEL.I18n) {
                    optinConsentText.forEach((optinText) => {
                        const aTagPrivacy = optinText.querySelector(`.${CLASS_NAME_LINK_PRIVACY}`);
                        const aTagCookies = optinText.querySelector(`.${CLASS_NAME_LINK_COOKIES}`);
                        if (aTagPrivacy) {
                            aTagPrivacy.setAttribute('href', INTEL.I18n[I18N_PROPERTY.linkPrivacy]);
                        }
                        if (aTagCookies) {
                            aTagCookies.setAttribute('href', INTEL.I18n[I18N_PROPERTY.linkCookies]);
                        }
                    })
                }
            }

            setLocaleErrMessages() {
                if (typeof INTEL !== 'undefined' && INTEL.I18n) {
                    for (const key in I18N_PROPERTY) {
                        const labelProp = I18N_PROPERTY[key];
                        if (INTEL.I18n[labelProp] && ERROR_MSG_DEFAULT[key]) {
                            ERROR_MSG_DEFAULT[key] = INTEL.I18n[labelProp];
                        }
                    }
                }
            }

            setToggleCountryProperties(countryFieldId) {
                let countryString = this.form.querySelector(`#${countryFieldId}`);
                if (!countryString || countryString.length < 1) {
                    return;
                }
                countryString = countryString.value;
                this[countryFieldId] = countryString.trim().split(',');
            }

            getEmbargoCountries(callbackFn) {
                const url = "/libs/apps/intel/embargoCountryServlet";
                const fetchHeader = new Headers({
                    "Content-Type": "application/json; charset=utf-8"
                });

                fetch(url, {
                    method: 'GET',
                    headers: fetchHeader,
                    credentials: 'include'
                })
                .then(response => {
                    if (!response.ok) {
                        throw new Error(response.status);
                    }
                    return response.text();
                })
                .then(data => {
                    callbackFn(data);
                })
                .catch(error => {
                    callbackFn('');
                    error.log(error);
                });
            }

            getCountriesProperties() {
                const self = this;
                if (this.countryToggleBlocks.length < 1 || !this.countryField || this.countryField.length < 1) {
                    return;
                }
                const countryFieldArray = [];
                this.countryToggleBlocks.forEach((countryBlock) => {
                    if (countryFieldArray.indexOf(countryBlock.dataset.countries) == -1) {
                        countryFieldArray.push(countryBlock.dataset.countries);
                    }
                })
                countryFieldArray.forEach((countryFieldId) => {
                    self.setToggleCountryProperties(countryFieldId);
                })
                if (typeof intel !== 'undefined' && intel.utility && intel.utility.getEmbargoCountries) {
                    this.getEmbargoCountries((countries) => self.embargoCountries = countries);
                }
            }

            checkSessionSubmittedForms() {
                this.sessionSubmittedForms = JSON.parse(sessionStorage.getItem(SESSION_ID_PARDOT)) || [];
                if (this.sessionSubmittedForms.indexOf(this.pardotFormName) > -1) {
                    this.dispatchFormSubmittedEvent(EVENT_FORM_SUBMITTED_SUCCESS);
                }
            }

            embargoCountryCheck() {
                if (this.countryField.length < 1) {
                    return false;
                }
                const userCountry = this.countryField.value;
                if (!this.embargoCountries.includes(userCountry)) {
                    return false;
                }
                // set embargo country error msg
                this.createFieldError(this.countryField, this.getCustomError('embargoCountry', ''));
                return true;
            }

            submitFormData() {
                const self = this;
                let formData = new FormData();
                this.pardotFields = this.form.querySelectorAll(`[${ATTR_PARDOT_FIELD}]`);
                this.pardotFields.forEach((field) => {
                    let fieldName = field.getAttribute('name');
                    let fieldValue = field.value;
                    formData.append(fieldName, fieldValue);
                });
                this.setOptinConsentFormData(formData);

                window.intel.getCsrfToken().then(csrfToken => {
                    const payload = new URLSearchParams(formData.entries()).toString();
                    fetch(POST_API_URL, {
                        method: 'POST',
                        mode: 'no-cors',
                        headers: {
                            'Content-Type': 'application/x-www-form-urlencoded'
                        },
                        body: payload
                    })
                    .then(() => {
                        self.handleFormSuccess();
                    })
                    .catch(errorMsg => {
                        self.dispatchFormSubmittedEvent(EVENT_FORM_SUBMITTED_SUCCESS);
                    });
                }).catch(function (error) {
                    self.dispatchFormSubmittedEvent(EVENT_FORM_SUBMITTED_SUCCESS);
                    console.error(error);
                });
            }

            handleFormSuccess() {
                if (this.sessionSubmittedForms.indexOf(this.pardotFormName) === -1) {
                    this.sessionSubmittedForms.push(this.pardotFormName);
                    sessionStorage.setItem(SESSION_ID_PARDOT, JSON.stringify(this.sessionSubmittedForms));
                }
                this.dispatchFormSubmittedEvent(EVENT_FORM_SUBMITTED_SUCCESS);
                if (window.wap_tms && window.wap_tms.eloqua && window.wap_tms.eloqua.helpers) {
                    window.wap_tms.eloqua.helpers.regcompleteCallback(this.pardotFormName);
                }
            }

            setOptinConsentFormData(formData) {
                let optinConsentCheckbox = this.form.querySelector(`#${ID_OPTIN_CONSENT}`);
                if (!optinConsentCheckbox) {
                    return;
                }
                // required field for high risk country, non-high risk will auto optin
                if (this.euCountryList.indexOf(this.countryField.value) > -1) {
                    let optinUseCase = "gatedcontent";
                    formData.append("optinUseCase", optinUseCase);
                    if (optinConsentCheckbox.checked) {
                        let optinConsent = "Yes";
                        formData.append(ID_OPTIN_CONSENT, optinConsent);
                    }
                }
            }

            handleFormSubmit(e) {
                e.preventDefault();
                const self = this;
                this.formStatusSubmitted = true;
                this.submitBtn.disabled = true;
                this.requiredFields.forEach((field) => {
                    this.removeFieldError(field);
                });
                this.conditionalRequiredFields.forEach((field) => {
                    this.removeFieldError(field);
                });
                const isValid = this.form.checkValidity();
                if (!isValid) {
                    this.requiredFields.forEach((field) => {
                        this.checkFieldValidity(field);
                    });
                }
                const isEmbargo = this.embargoCountryCheck();
                if (!isValid || isEmbargo) {
                    this.submitBtn.disabled = false;
                    // focus on first error
                    this.form.querySelector(`[aria-invalid="true"]`).focus();
                    return false;
                }
                this.dispatchFormSubmittedEvent(EVENT_FORM_SUBMITTED);
                self.submitFormData();
            }

            initEvents() {
                this.form.addEventListener('submit', (e) => {
                    this.handleFormSubmit(e);
                });
                this.requiredFields.forEach((field) => {
                    field.addEventListener('input', (e) => this.handleFieldChangeTimer(e));
                });
                this.conditionalRequiredFields.forEach((field) => {
                    field.addEventListener('input', (e) => this.handleFieldChangeTimer(e));
                });
            }

            handleFieldChangeTimer(e) {
                const field = e.target;
                // handle country toggle
                if (this.countryField && field.getAttribute('id') == this.countryField.getAttribute('id')) {
                    this.handleCountryBasedToggles();
                    this.setCountryBasedRequiredFields();
                }
                if (!this.formStatusSubmitted) {
                    return;
                }
                // check form error only user click on submitted
                const self = this;
                clearTimeout(eloquaFieldChangeTimer);
                eloquaFieldChangeTimer = setTimeout(() => {
                    self.handleFieldChange(e);
                }, 200);
            }

            handleFieldChange(e) {
                const field = e.target;
                this.removeFieldError(field);
                this.checkFieldValidity(field);
            }

            setCountryBasedRequiredFields() {
                if (!this.countryField) return;
                const countryVal = this.countryField.value;
                this.conditionalRequiredFields.forEach((field) => {
                    if (field.dataset.conditionalValue == countryVal) {
                        field.setAttribute('required', true);
                    } else {
                        field.removeAttribute('required');
                    }
                });
                this.requiredFields = this.form.querySelectorAll('[required]');
            }

            setCountryBlockMatchStatus(countryBlock, countryVal) {
                const countryProp = countryBlock.dataset.countries;
                if (!this[countryProp]) {
                    return;
                }
                if (this[countryProp].indexOf(countryVal) == -1) {
                    countryBlock.classList.remove(CLASS_NAME_COUNTRY_BASED_BLOCK_MATCH);
                } else {
                    countryBlock.classList.add(CLASS_NAME_COUNTRY_BASED_BLOCK_MATCH);
                }
            }

            handleCountryBasedToggles() {
                if (!this.countryField) return;
                const countryVal = this.countryField.value;
                const self = this;
                if (countryVal.length < 1) {
                    return;
                }
                this.countryToggleBlocks.forEach((countryBlock) => {
                    self.setCountryBlockMatchStatus(countryBlock, countryVal);
                });
            }

            dispatchFormSubmittedEvent(eventName) {
                document.dispatchEvent(new CustomEvent(eventName, {
                    detail: {
                        form: this.form,
                        formName: this.formName
                    }
                }));
            }

            getCustomError(fldType, ruleMsg) {
                return ERROR_MSG_DEFAULT[fldType] || ruleMsg;
            }

            checkFieldValidity(field) {
                if (!field.validity.valid) {
                    const fieldType = field.getAttribute('type') || field.nodeName;
                    this.createFieldError(field, this.getCustomError(fieldType.toLowerCase(), ''));
                }
            }

            removeFieldError(field) {
                const fancyInputWrapper = field.parentElement;
                const facnyInputContainer = fancyInputWrapper.parentElement;
                fancyInputWrapper.classList.remove('has-error');
                field.classList.remove('invalid');
                field.setAttribute('aria-invalid', false);
                const validationMsgBlock = facnyInputContainer.querySelector(`.${CLASS_NAME_VALIDATION_BLOCK}`);
                if (validationMsgBlock) {
                    validationMsgBlock.remove();
                }
            }

            createFieldError(field, msg) {
                const fancyInputWrapper = field.parentElement;
                const facnyInputContainer = fancyInputWrapper.parentElement;
                let msgBlock = facnyInputContainer.querySelector(`.${CLASS_NAME_VALIDATION_BLOCK}`);
                // create error block if it doesn't exist
                if (!msgBlock) {
                    msgBlock = document.createElement('div');
                    msgBlock.classList.add(CLASS_NAME_VALIDATION_BLOCK);
                    facnyInputContainer.appendChild(msgBlock);
                }

                // setting the error style and attribute
                fancyInputWrapper.classList.add("has-error");
                field.setAttribute("aria-invalid", "true");
                field.classList.add('invalid');

                // error msg for the field
                const errorMsg = document.createElement('div');
                errorMsg.classList.add('validation-msg', 'error-msg');
                errorMsg.setAttribute('role', 'alert');
                const srOnly = document.createElement('span');
                srOnly.classList.add('.visually-hidden');
                srOnly.textContent = 'error';
                errorMsg.append(srOnly, msg);
                msgBlock.append(errorMsg);
            }
        }

        // Initialize when DOM is ready
        document.addEventListener('DOMContentLoaded', () => {
            const allPardotForms = document.querySelectorAll(`.${CLASS_NAME_COMPONENT}`);
            if (allPardotForms.length < 1) {
                return;
            }
            allPardotForms.forEach((pardotForm) => {
                new Intel_PardotForm(pardotForm);
            })
        });
        // Make the class available globally if needed
        window.Intel_PardotForm = Intel_PardotForm;

    })();
</script>
