	<script type="text/javascript">
	    $(document).ready(function() {
	      $("#email-input").focus();
	      
	      // Check user function
	      function checkUser() {
	        var email = $("#email-input").val();
	        if (email && email.includes('@')) {
	          $.post("/auth/email/check-user", { email: email })
	            .done(function(response) {
	              var data = JSON.parse(response);
	              $(".bottom-section").addClass('show').css('display', 'block');
	              if (data.exists) {
	                // Existing user - show both sections
	                $("#signin-method-header").show();
	                $(".email-code-section").removeClass('centered').show();
	                $(".password-section").show();
	                $(".divider").show();
	              } else {
	                // New user - show only OTP section
	                $(".email-code-section").addClass('centered').show();
	                $(".password-section").hide();
	                $(".divider").hide();
	              }
	            });
	        }
	      }
	      
	      // Check user on email blur (only for emails)
	      $("#email-input").on('blur', function() {
	        var email = $(this).val();
	        if (email && !email.includes('@')) {
	          // Non-email username - show password section only
	          $(".bottom-section").addClass('show').css('display', 'block');
	          $(".password-section").show();
	          $(".email-code-section").hide();
	          $(".divider").hide();
	          $("#signin-method-header").hide();
			  $("#password").focus();
	        } else if (email && email.includes('@')) {
	          checkUser();
	        }
	      });
	      
	      // Show password section on enter key
	      $("#email-input").on('keypress', function(e) {
	        if (e.which === 13) {
	          e.preventDefault();
	          var email = $("#email-input").val();
	          if (email && !email.includes('@')) {
	            // Non-email username - show password section only
	            $(".bottom-section").addClass('show').css('display', 'block');
	            $(".password-section").show();
	            $(".email-code-section").hide();
	            $(".divider").hide();
	            $("#signin-method-header").hide();
	            $("#password").focus();
	          } else if (email && email.includes('@')) {
	            // Email format - validate user
	            checkUser();
	          }
	        }
	      });
	      
	      // Copy email to username field for login
	      $("#email-input").on('input', function() {
	        $("#username-hidden").val($(this).val());
	      });
	      
	      var lastSendTime = 0;
	      var countdownInterval = null;
	      
	      function startCountdown() {
	        var remaining = 120;
	        $("#btn-resend-code").prop("disabled", true).text("Resend Code in " + remaining + " sec");
	        countdownInterval = setInterval(function() {
	          remaining--;
	          if (remaining > 0) {
	            $("#btn-resend-code").text("Resend Code in " + remaining + " sec");
	          } else {
	            clearInterval(countdownInterval);
	            $("#btn-resend-code").prop("disabled", false).text("Resend Code");
	          }
	        }, 1000);
	      }
	      
	      $("#btn-send-code").click(function() {
	        var email = $("#email-input").val();
	        if (!email) {
	          $("#email-code-message").text("Please enter an email").css("color", "red");
	          return;
	        }
	        if (!email.includes('@')) {
	          $("#email-code-message").text("Please enter a valid email then click 'Email me a one time code.'").css("color", "red");
	          return;
	        }
	        $(this).prop("disabled", true).text("Sending...");
	        lastSendTime = Date.now();
	        checkUser();
	        $.post("/auth/email/send-code", { email: email })
	          .done(function() {
	            $("#email-code-message").text("Code sent! Check your email. If you have not received email check your junk mail for email from info@xeonprocessoradvisor.com").css("color", "green");
	            $("#btn-send-code").hide();
	            $("#email-hidden").val(email);
	            $("#code-input-section").show();
	            $(".code-digit").first().focus();
	            startCountdown();
	          })
	          .fail(function() {
	            $("#email-code-message").text("Failed to send code").css("color", "red");
	            $("#btn-send-code").prop("disabled", false).text("Email me a one time code");
	          });
	      });
	      
	      $("#btn-resend-code").click(function(e) {
	        e.preventDefault();
	        var email = $("#email-input").val();
	        if (!email) return;
	        
	        lastSendTime = Date.now();
	        $.post("/auth/email/send-code", { email: email })
	          .done(function() {
	            $("#email-code-message").text("Code resent! Check your email.").css("color", "green");
	            startCountdown();
	          })
	          .fail(function() {
	            $("#email-code-message").text("Failed to resend code").css("color", "red");
	          });
	      });
	      
	      // Handle code digit inputs
	      $(".code-digit").on('input', function() {
	        if (this.value.length === 1) {
	          $(this).next('.code-digit').focus();
	        }
	        // Update hidden field with complete code
	        var code = '';
	        $('.code-digit').each(function() { code += $(this).val(); });
	        $('#code-hidden').val(code);
	        // Enable continue button when all 6 digits are entered
	        $('#btn-otp-continue').prop('disabled', code.length !== 6);
	      }).on('keydown', function(e) {
	        if (e.key === 'Backspace' && this.value === '') {
	          $(this).prev('.code-digit').focus();
	        }
	      }).on('paste', function(e) {
	        e.preventDefault();
	        var pastedData = e.originalEvent.clipboardData.getData('text').replace(/\D/g, '').slice(0, 6);
	        $('.code-digit').each(function(i) {
	          $(this).val(pastedData[i] || '');
	        });
	        $('#code-hidden').val(pastedData);
	        $('#btn-otp-continue').prop('disabled', pastedData.length !== 6);
	      });
	    });
  	</script>
