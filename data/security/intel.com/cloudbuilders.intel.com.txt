function createGA4Cookies() {
    // Load GA4 Library
    var gaScript = document.createElement('script');
    gaScript.async = true;
    gaScript.src = 'https://www.googletagmanager.com/gtag/js?id=G-TRJS24038B';
    document.head.appendChild(gaScript);

    // Initialize GA4
    window.dataLayer = window.dataLayer || [];
    function gtag() { dataLayer.push(arguments); }
    window.gtag = gtag;

    gtag('js', new Date());

    // Configure GA4
    gtag('config', 'G-TRJS24038B', {
        client_storage: 'cookie',
        cookie_domain: 'auto',
        cookie_expires: 63072000,
        cookie_update: true,
        // Optional: Send clean URL
        page_path: window.location.pathname
    });

    // Fire page_view manually (if needed)
    gtag('event', 'page_view', {
        page_location: window.location.href,
        page_path: window.location.pathname,
        page_title: document.title
    });

    // Create GA cookies if missing
    if (!document.cookie.includes("_ga")) {
        let gaClientId = "GA1.1." + Math.floor(Math.random() * 1000000000) + "." + Math.floor(Date.now() / 1000);
        document.cookie = "_ga=" + gaClientId + "; path=/; domain=" + window.location.hostname + "; expires=" + new Date(Date.now() + 63072000000).toUTCString() + "; Secure; SameSite=None";
    }

    if (!document.cookie.includes("_gid")) {
        let gidClientId = "GA1.1." + Math.floor(Math.random() * 1000000000) + "." + Math.floor(Date.now() / 1000);
        document.cookie = "_gid=" + gidClientId + "; path=/; domain=" + window.location.hostname + "; expires=" + new Date(Date.now() + 86400000).toUTCString() + "; Secure; SameSite=None";
    }
}

function loadTwitterIframeExecute() {

        return;
    }

    // Check if already in viewport
    function isInViewport(el) {
        const rect = el.getBoundingClientRect();
        return (
            rect.top < window.innerHeight &&
            rect.bottom >= 0
        );
    }

    if (isInViewport(twitterContainer)) {
        loadTwitterIframe(); // load immediately
    }

    window.addEventListener('scroll', function () {
        loadTwitterIframe(); // fallback for scroll
    });
}


function loadRecaptchaAndExecute() {
    
    
        if (window.recaptchaAlreadyLoaded) return;
        window.recaptchaAlreadyLoaded = true;
        
        // Check if the #g-recaptcha-response element exists
    if (!document.getElementById('g-recaptcha-response')) {
        console.log("Element with ID 'g-recaptcha-response' not found. Skipping reCAPTCHA.");
        return;
    }

        var scriptTag = document.createElement('script');
        scriptTag.src = 'https://www.google.com/recaptcha/api.js?render=6LcJFQopAAAAAClOZlyXnkw2Sq_c4wPvt6gDGQ02';
        scriptTag.async = true;
        scriptTag.defer = true;
        scriptTag.type = 'text/javascript';
        
        
       

        scriptTag.onload = function () {
            console.log("reCAPTCHA script loaded into head ✅");
            executeRecaptcha();
        };

        scriptTag.onerror = function () {
            console.error("Failed to load reCAPTCHA script ❌");
        };

        // Correct way: Append inside HEAD properly
        var head = document.getElementsByTagName('head')[0];
        if (head) {
            head.appendChild(scriptTag);
        } else {
            // fallback if <head> not found
            console.error("No <head> tag found!");
            document.body.appendChild(scriptTag);
        }
    }

    function executeRecaptcha() {
        if (typeof grecaptcha === 'undefined') {
            console.error("grecaptcha is not available yet ❌");
            return;
        }

        grecaptcha.ready(function () {
            grecaptcha.execute('6LcJFQopAAAAAClOZlyXnkw2Sq_c4wPvt6gDGQ02', {
                action: 'emailSubscribed'
            }).then(function (joomlakey) {
                //console.log("Initial token received ✅:", joomlakey);
                document.getElementById('g-recaptcha-response').value = joomlakey;

                // Refresh token every 90 seconds
                setInterval(function () {
                    grecaptcha.execute('6LcJFQopAAAAAClOZlyXnkw2Sq_c4wPvt6gDGQ02', {
                        action: 'emailSubscribed'
                    }).then(function (newjoomlakey) {
                        document.getElementById('g-recaptcha-response').value = newjoomlakey;
                        //console.log("Refreshed token ✅:", newjoomlakey);
                    });
                }, 90 * 1000);
            });
        });
    }

<script>
(function () {
  const isHomePage =
    window.location.pathname === "/" ||
    window.location.pathname === "/index.html";

  if (!isHomePage) return;

  // 🔥 Remove all selected
  const removeSelected = () => {
    document
      .querySelectorAll(".nav-link-primary.selected")
      .forEach((el) => el.classList.remove("selected"));
  };

  // 🔥 Remove all focused (NEW)
  const removeFocused = () => {
    document
      .querySelectorAll(".nav-link-primary.focused")
      .forEach((el) => el.classList.remove("focused"));
  };

  // 🚀 Initial cleanup
  removeSelected();
  document.addEventListener("DOMContentLoaded", removeSelected);
  window.addEventListener("load", removeSelected);

  // 🚀 MutationObserver
  const observer = new MutationObserver(() => {
    removeSelected();
  });

  observer.observe(document.documentElement, {
    subtree: true,
    attributes: true,
    attributeFilter: ["class"],
  });

  // ✅ 🔥 CLICK BEHAVIOR (UPDATED)
  document.addEventListener("click", function (e) {
    const target = e.target.closest(".nav-link-primary");
    const navContainer = document.getElementById("primary-nav");

    // 👉 Agar nav ke bahar click hua → focused remove
    if (navContainer && !navContainer.contains(e.target)) {
      removeFocused();
      return;
    }

    // 👉 Agar link pe click hua
    if (target) {
      removeFocused();
      target.classList.add("focused");
    }
  });
})();
</script>

function disableAnchorById(anchorId) {
    var anchor = jQuery("#" + anchorId);
    anchor.removeAttr("href");
    anchor.css("pointer-events", "none");
    anchor.css("color", "gray");
}

    var anchor = jQuery("#" + anchorId);
    anchor.attr("href", "#");
    anchor.css("pointer-events", "");
    anchor.css("color", "");

function ValidateEmail(input) {
    var validRegex = /^\w+([\.-]?\w+)*@\w+([\.-]?\w+)*(\.\w{2,3})+$/;

    if (input.match(validRegex)) {

        return true;

        return false;

    }

}

        var regex = new RegExp("^[a-zA-Z ]");

        if (regex.test(str)) {
            return true;

        }
    });

// Function to handle intersection observer
function lazyLoadSection(entries, observer) {
    entries.forEach((entry) => {
        if (entry.isIntersecting) {
            entry.target.classList.add("visible"); // Add 'visible' class to make the section visible
            observer.unobserve(entry.target); // Stop observing once it's visible
        }
    });
}

// Create an intersection observer
const lazyLoadObserver = new IntersectionObserver(lazyLoadSection, {
    root: null,
    rootMargin: "0px",
    threshold: 0.1, // 10% of the section needs to be visible
});

// Target the section to be lazily loaded
const lazyLoadSections = document.querySelectorAll(".lazy-load");

// Observe each section
lazyLoadSections.forEach((section) => {
    lazyLoadObserver.observe(section);
});
