    /**
     * Determine the mobile operating system.
     * This function returns one of 'iOS', 'Android', 'Windows Phone', or 'unknown'.
     *
     * @returns {String}
     */
    function getMobileOperatingSystem() {
        var userAgent = navigator.userAgent || navigator.vendor || window.opera;

        // Windows Phone must come first because its UA also contains "Android"
        if (/windows phone/i.test(userAgent)) return "WindowsPhone";
        if (/android/i.test(userAgent)) return "Android";
        // iOS detection from: http://stackoverflow.com/a/9039885/177710
        if (/iPad|iPhone|iPod/.test(userAgent) && !window.MSStream) return "iOS";

        return "unknown";
    }

    /**
     * Determine the operating system.
     * This function returns one of 'Windows', 'MacOS', 'UNIX', 'Linux' or 'unknown'.
     *
     * @returns {String}
     */
    function getOperatingSystem() {
        if (navigator.appVersion.indexOf("Win") != -1) return "Windows";
        if (navigator.appVersion.indexOf("Mac") != -1) return "MacOS";
        if (navigator.appVersion.indexOf("X11") != -1) return "UNIX";
        if (navigator.appVersion.indexOf("Linux") != -1) return "Linux";

        return "unknown";
    }

    /**
     * set conversation variables
     *
     * @param {object} cmwcInstance
     */
    function setConversationContext(cmwcInstance) {

        if(!window.conversationContext)
        {
            window.conversationContext = {
                geoIP                : 'US',
                country              : '',
                websiteLanguage      : 'en',
                mobileOperatingSystem: getMobileOperatingSystem(),
                operatingSystem      : getOperatingSystem(),
                
                            };
        }

        // Add Tracedock to agent context asynchronously. 
        // It will be present in future contexts if not in time for the first call.
        window.cookieStore.get("anonymous_id").then(cookie => {
            if (cookie && cookie.value) {
                window.conversationContext.anonymous_id = cookie.value;
                console.log("Tracedock id added to context");
            }
        }).catch(e => {
            console.error("Error fetching tracedock id", e);
        });

        const storedData = sessionStorage.getItem('formData');
        const formData = (storedData) ? JSON.parse(storedData) : {};

        const context = Object.assign({}, window.conversationContext, formData);

        for (const i in context) {
            if (typeof context[i] === 'undefined' || context[i] === '') {
                delete context[i];
            }
        }        

        console.log("Conversation context set: ", context);
        cmwcInstance.addContext(context);

    }
