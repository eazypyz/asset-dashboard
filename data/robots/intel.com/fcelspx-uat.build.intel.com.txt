  <!-- Bundle version check script - keeps this inline for fail-safe version checking -->
  <!-- This script ensures the correct bundle version is loaded after deployments -->
  <script>
    (function() {
      const MAX_REFRESHES = 3;
      const BUNDLE_VERSION_REFRESH_KEY = 'bundleVersionRefreshCount';
      
      function getRefreshCount() {
        return parseInt(sessionStorage.getItem(BUNDLE_VERSION_REFRESH_KEY) || '0');
      }

      function incrementRefreshCount() {
        const count = getRefreshCount() + 1;
        sessionStorage.setItem(BUNDLE_VERSION_REFRESH_KEY, count.toString());
        return count;
      }

      function forceReload() {
        const refreshCount = incrementRefreshCount();
        if (refreshCount <= MAX_REFRESHES) {
          console.log(`Bundle version mismatch - forcing reload (attempt ${refreshCount}/${MAX_REFRESHES})`);
          window.location.reload(true);
        } else {
          console.log(`Max version refresh attempts (${MAX_REFRESHES}) reached - stopping version check`);
        }
      }

      function findMainBundlePath() {
        try {
          const scripts = document.getElementsByTagName('script');
          for (const script of scripts) {
            const src = script.getAttribute('src');
            if (src && (src === 'main.js' || src.match(/main[-.]?[A-Z0-9]+(\.bundle)?\.js$/))) {
              console.log('Found bundle:', src);
              return src;
            }
          }
          console.log('Bundle not found in current scripts');
          return null;
        } catch (error) {
          console.error('Error finding bundle:', error);
          return null;
        }
      }

      function checkBundleVersion(retryCount = 0) {
        console.log('Checking bundle version...');
        const mainBundlePath = findMainBundlePath();
        if (!mainBundlePath) {
          if (retryCount < 5) {
            console.log(`Bundle not loaded yet, retrying in 1 second... (attempt ${retryCount + 1}/5)`);
            setTimeout(() => checkBundleVersion(retryCount + 1), 1000);
          } else {
            console.log('Max retries reached - refreshing for new version');
            forceReload();
          }
          return;
        }

        fetch(mainBundlePath, { cache: 'no-store' })
          .then(response => {
            const contentType = response.headers.get('content-type');
            if (!contentType || !contentType.includes('javascript')) {
              console.log('Bundle version mismatch: server returned non-JavaScript content');
              forceReload();
              return;
            }
            console.log('Bundle version verified successfully');
            sessionStorage.setItem(BUNDLE_VERSION_REFRESH_KEY, '0');
          })
          .catch(error => {
            console.log('Error checking bundle version:', error);
            forceReload();
          });
      }

      console.log('Starting bundle version check system');
      checkBundleVersion();

      document.addEventListener('visibilitychange', function() {
        if (document.visibilityState === 'visible') {
          console.log('Tab became visible - checking bundle version');
          checkBundleVersion();
        }
      });
    })();
  </script>
  <!-- End bundle version check script -->
