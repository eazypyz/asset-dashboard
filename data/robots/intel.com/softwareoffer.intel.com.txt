                var innerContainerHeightDifference = lowestDescriptionBottom - innerContainerPaddingTop - originalInnerContainerHeight; 
                if (innerContainerHeightDifference != 0) {
                    var height = originalInnerContainerHeight + innerContainerHeightDifference;
                    $divFeaturetteContentInnerContainers.height(height);
                    $divFeaturetteContentOuterContainers.outerHeight($divFeaturetteContentInnerContainers.first().outerHeight())
                    $divFeaturetteContentOuterContainers.css('padding-bottom', 'unset');
                }
            }
        }

            $(window).resize(onWindowResize);

        function initRotator() {
            $banners.fadeOut(0);
            $firstBanner.fadeIn(0).addClass(classActive);

            if ($banners.length > 1) {
                $btnBannerNavigation.show();
                startRotator();
                $btnPrev.click({ prev: true }, btnBannerNavigationClicked);
                $btnNext.click({ prev: false }, btnBannerNavigationClicked);
            } else {
                $btnBannerNavigation.hide();
            }
        }

        function btnBannerNavigationClicked(event) {
            stopRotator();
            rotateBanners(event.data.prev);
            startRotator();
        }

        function startRotator() {
            rotatorInterval = setInterval(rotateBanners, intervalRotateMS);
        }

        function rotateBanners(prev) {
            $activeBanner = $(activeBanner);
            var $nextBanner;
            if (prev) {
                $nextBanner = $activeBanner.prev(selectorDiv);
                if ($nextBanner.length == 0) {
                    $nextBanner = $lastBanner;
                }
            } else {
                $nextBanner = $activeBanner.next(selectorDiv);
                if ($nextBanner.length == 0) {
                    $nextBanner = $firstBanner;
                }
            }
            $activeBanner.fadeOut(intervalFadeMS).removeClass(classActive);
            $nextBanner.fadeIn(intervalFadeMS).addClass(classActive);
        }

        function stopRotator() {
            clearInterval(rotatorInterval);
        }
