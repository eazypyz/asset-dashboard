<script>
    const scripts = [
        'https://www.cm.com/app/aurora/js/cm-date-time-service.js',
        'https://www.cm.com/app/aurora/js/webcomponents-loader.js',
        'https://www.cm.com/app/aurora/js/aurora-components-legacy.js'
    ];
    scripts.forEach(src => {
        const script = document.createElement('script');
        script.src = src;
        document.body.appendChild(script);
    });
</script>


<script>
    
    customElements.whenDefined('cm-date-time-picker').then(() => {
        document.getElementById('date-picker').addEventListener('change', (e) => {
            document.getElementById('date-input').value = e.detail.date;
            state.hasDate = true;
            updateResendButton();
        });
    });

    
    const state = {
        hasValidOrderId: false,
        hasValidEmail: false,
        hasDate: false,
        hasRecaptcha: false
    };

    
    function onRecaptchaSuccess(token) {
        state.hasRecaptcha = true;
        if (document.getElementById('email-input') && document.getElementById('date-picker')) {
            updateResendButton();
        } else if (document.getElementById('order-id-input')) {
            updateResaleButton();
        }
    }

    
    function updateResaleButton() {
        const sendButton = document.getElementById('send-button');
        if (state.hasValidOrderId && state.hasRecaptcha) {
            sendButton.removeAttribute('disabled');
        } else {
            sendButton.setAttribute('disabled', 'true');
        }
    }

    function updateResendButton() {
        const sendButton = document.getElementById('send-button');
        if (state.hasValidEmail && state.hasDate && state.hasRecaptcha) {
            sendButton.removeAttribute('disabled');
        } else {
            sendButton.setAttribute('disabled', 'true');
        }
    }

    
    const emailInput = document.getElementById('email-input');
    const emailError = document.getElementById('email-error');
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

    if (emailInput) {
        emailInput.addEventListener('input', e => {
            const email = e.target.value.trim();
            const valid = email !== '' && emailRegex.test(email);

            if (email !== '' && !valid) {
                e.target.setAttribute('data-error', 'true');
                emailError.classList.add('show');
            } else {
                e.target.removeAttribute('data-error');
                emailError.classList.remove('show');
            }

            state.hasValidEmail = valid;
            updateResendButton();
        });
    }

    
    const orderIdInput = document.getElementById('order-id-input');
    const orderIdError = document.getElementById('order-id-error');
    const orderIdRegex = /^[A-Z]{2}.{8}$/;

    if(orderIdInput) {
        orderIdInput.addEventListener('input', e => {
            const orderId = e.target.value.trim();
            const valid = orderId !== '' && orderIdRegex.test(orderId);

            if (orderId !== '' && !valid) {
                e.target.setAttribute('data-error', 'true');
                orderIdError.classList.add('show');
            } else {
                e.target.removeAttribute('data-error');
                orderIdError.classList.remove('show');
            }

            state.hasValidOrderId = valid;
            updateResaleButton();
        })
    }

    
    customElements.whenDefined('cm-select').then(() => {
        const languageSelect = document.getElementById('language-select');

        
        const currentLanguage = 'EN';
        if (currentLanguage) {
            languageSelect.value = currentLanguage;
        }

        
        languageSelect.addEventListener('change', (e) => {
            const selectedLanguage = e.detail.value.toLowerCase();

            const isResalePage = window.location.pathname.startsWith('/resale');
            const basePath = isResalePage ? '/resale/' : '/';
            window.location.href = basePath + selectedLanguage;
        });
    });
</script>
