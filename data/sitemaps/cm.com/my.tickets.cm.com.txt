    function showToast(level, message) {
        message = (message || "").trim();
        const root = document.getElementById("cm-toasts");
        if (!message || !root) return;
        const styles = {
            success: "bg-gn-10 border-gn-100 text-gn-160",
            info: "bg-cm-soft border-cm text-cm-deep",
        };
        const el = document.createElement("div");
        el.setAttribute("role", "alert");
        el.className = "pointer-events-auto w-full max-w-[380px] flex items-start gap-2.5 rounded-xl border px-4 py-3 text-[13px] font-bold shadow-elevated transition-opacity duration-300 " +
            (styles[level] || styles.info);
        el.textContent = message;
        el.addEventListener("click", () => el.remove());
        root.appendChild(el);
        setTimeout(() => { el.style.opacity = "0"; setTimeout(() => el.remove(), 300); }, 4000);
    }

    document.body.addEventListener("makeToastReload", (e) => {
        try { sessionStorage.setItem("cm:toast", JSON.stringify(e.detail)); } catch (err) {}
        location.reload();
    });
    (function () {
        try {
            const raw = sessionStorage.getItem("cm:toast");
            if (raw) { sessionStorage.removeItem("cm:toast"); const t = JSON.parse(raw); showToast(t.level, t.message); }
        } catch (e) {}
    })();
