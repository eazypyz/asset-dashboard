    <script hx-preserve="true">
        window.showConfirmDialog = function (message) {
            return new Promise((resolve) => {
                const dialog = document.getElementById('htmx-confirm-dialog');
                const messageEl = document.getElementById('htmx-confirm-message');
                const cancelBtn = document.getElementById('htmx-confirm-cancel');
                const okBtn = document.getElementById('htmx-confirm-ok');

                messageEl.textContent = message;

                const controller = new AbortController();

                let resolved = false;

                const cleanup = (result) => {
                    if (resolved) return;
                    resolved = true;
                    controller.abort();
                    resolve(result);
                };

                cancelBtn.addEventListener('click', () => {
                    dialog.close();
                }, {signal: controller.signal, once: true});

                okBtn.addEventListener('click', () => {
                    cleanup(true);
                    dialog.close();
                }, {signal: controller.signal, once: true});

                dialog.addEventListener('cm-dialog-closed', () => {
                    cleanup(false);
                }, {signal: controller.signal, once: true});

                dialog.open();
            });
        };

        document.body.addEventListener('htmx:confirm', function (event) {
            if (!event.detail.question) {
                return;
            }

            event.preventDefault();

            window.showConfirmDialog(event.detail.question).then((confirmed) => {
                if (confirmed) {
                    event.detail.issueRequest(true);
                }
            });
        });
    </script>
