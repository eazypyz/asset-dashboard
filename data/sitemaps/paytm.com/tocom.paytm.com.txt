<script>
function getAdsManagerHostPublic() {
    const currentHost = window.location.hostname;
    if (currentHost.includes('tocom-ui-tocom-qa-internal.mypaytm.com')) {
        return 'ads-manager-qa-internal.mypaytm.com';
    } else if (currentHost.includes('tocom-ui-tocom-qa.paytm.com')) {
        return 'adsmanager-qa.paytm.com';
    } else if (currentHost.includes('tocom.paytm.com')) {
        return 'adsmanager.paytm.com';
    } else {
        return 'ads-manager-qa-internal.mypaytm.com';
    }
}

function redirectToAdsManagerHome() {
    const adsManagerHost = getAdsManagerHostPublic();
    const adsManagerUrl = "https://" + adsManagerHost;
    window.open(adsManagerUrl, 'myWindow');
}

function redirectToAdsManagerRequestDemo() {
    const adsManagerHost = getAdsManagerHostPublic();
    const adsManagerUrl = "https://" + adsManagerHost + "/request-demo";
    window.open(adsManagerUrl, 'myWindow');
}
